import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();

        int people = 0;
        int max = 0;

        for (int i = 0; i < n; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();

            people = people - a + b;

            max = Math.max(max, people);
        }

        System.out.println(max);
    }
}
