question


You have a fence consisting of n boards. The i-th board has an initial height \(a_{i}\). You want to make the fence "great." A fence is considered great if no two adjacent boards have the same height (i.e., \(a_i \neq a_{i+1}\) for all 1 ≤ i < n).


code


#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const long long INF = 1e18;

void solve() {
    int n;
    cin >> n;
    
    vector<long long> a(n), b(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i] >> b[i];
    }
    
    vector<vector<long long>> dp(n, vector<long long>(3, INF));
    
    dp[0][0] = 0;
    dp[0][1] = b[0];
    dp[0][2] = 2 * b[0];
    
    for (int i = 1; i < n; ++i) {
        for (int j = 0; j < 3; ++j) {
            for (int k = 0; k < 3; ++k) {
                if (a[i - 1] + j != a[i] + k) {
                    dp[i][k] = min(dp[i][k], dp[i - 1][j] + k * b[i]);
                }
            }
        }
    }
    
    long long ans = min({dp[n - 1][0], dp[n - 1][1], dp[n - 1][2]});
    cout << ans << "\n";
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int q;
    cin >> q;
    while (q--) {
        solve();
    }
    return 0;
}
