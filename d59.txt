code


#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int m, s;
    cin >> m >> s;

    if (s == 0) {
        if (m == 1) {
            cout << "0 0\n";
        } else {
            cout << "-1 -1\n";
        }
        return 0;
    }

    if (s > 9 * m) {
        cout << "-1 -1\n";
        return 0;
    }

    string max_num = "";
    int temp_s = s;
    for (int i = 0; i < m; i++) {
        int digit = min(9, temp_s);
        max_num += to_string(digit);
        temp_s -= digit;
    }

    string min_num = "";
    temp_s = s - 1;
    
    for (int i = 0; i < m - 1; i++) {
        int digit = min(9, temp_s);
        min_num += to_string(digit);
        temp_s -= digit;
    }
    
    min_num += to_string(temp_s + 1);
    reverse(min_num.begin(), min_num.end());

    cout << min_num << " " << max_num << "\n";

    return 0;
}
