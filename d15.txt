import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int k = sc.nextInt();
        int n = sc.nextInt();
        int w = sc.nextInt();

        int totalCost = 0;

        for (int i = 1; i <= n; i++) {
            totalCost += i * k;
        }

        if (totalCost > w) {
            System.out.println(totalCost - w);
        } else {
            System.out.println(0);
        }
    }
}
