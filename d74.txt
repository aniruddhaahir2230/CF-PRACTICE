Problem: Codeforces 1881E - Block Sequence

code

#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

void solve() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    // dp[i] stores the min deletions needed for the suffix starting at index i
    // dp[n] means we reached the end of the array successfully (0 deletions needed)
    vector<int> dp(n + 1, 0);

    for (int i = n - 1; i >= 0; i--) {
        // Option 1: Delete the current element
        dp[i] = 1 + dp[i + 1];

        // Option 2: Keep the current element as a block starter (if it fits)
        if (i + a[i] < n) {
            dp[i] = min(dp[i], dp[i + a[i] + 1]);
        }
    }

    cout << dp[0] << "\n";
}

int main() {
    // Fast I/O
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int t;
    cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
