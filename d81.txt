Problem: Cut Ribbon (189A)

#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    // Optimize standard I/O operations for performance
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n, a, b, c;
    cin >> n >> a >> b >> c;

    // dp[i] will store the maximum number of pieces for length i
    // Initialize with -1 to indicate an unreachable/invalid state
    vector<int> dp(n + 1, -1);
    
    // Base case: a ribbon of length 0 can be cut into exactly 0 pieces
    dp[0] = 0;

    // Compute maximum pieces for all lengths from 1 to n
    for (int i = 1; i <= n; ++i) {
        int choice1 = -1, choice2 = -1, choice3 = -1;

        if (i >= a && dp[i - a] != -1) choice1 = dp[i - a] + 1;
        if (i >= b && dp[i - b] != -1) choice2 = dp[i - b] + 1;
        if (i >= c && dp[i - c] != -1) choice3 = dp[i - c] + 1;

        dp[i] = max({choice1, choice2, choice3});
    }

    // Output the maximum pieces possible for total length n
    cout << dp[n] << "\n";

    return 0;
}
