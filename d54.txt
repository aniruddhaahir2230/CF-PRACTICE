#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;

    int mishka = 0, chris = 0;

    while (n--) {
        int x, y;
        cin >> x >> y;

        if (x > y)
            mishka++;
        else if (x < y)
            chris++;
    }

    if (mishka > chris)
        cout << "Mishka\n";
    else if (chris > mishka)
        cout << "Chris\n";
    else
        cout << "Friendship is magic!\n";

    return 0;
}
