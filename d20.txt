import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();

        int[] before = new int[n];
        int[] after = new int[n];

        boolean rated = false;
        boolean sorted = true;

        for (int i = 0; i < n; i++) {
            before[i] = sc.nextInt();
            after[i] = sc.nextInt();

            if (before[i] != after[i]) {
                rated = true;
            }

            if (i > 0 && before[i] > before[i - 1]) {
                sorted = false;
            }
        }

        if (rated) {
            System.out.println("rated");
        } else if (!sorted) {
            System.out.println("unrated");
        } else {
            System.out.println("maybe");
        }
    }
}
