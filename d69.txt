Problem Description

You are given an array a consisting of n positive integers, and an integer k. In one move, you can:
1. Choose any index i (1 ≤ i ≤ n) and increase \(a[i]\) by x (\(a[i] = a[i] + x\)).
2. After doing this, x must be increased by 1 (x = x + 1).
You can also choose to do nothing in a move, but x still increases by 1. Initially, x = 0. Your goal is to make all elements in the array divisible by k.
Find the minimum number of moves required to achieve this.



code


#include <iostream>
#include <vector>
#include <map>
#include <algorithm>

using namespace std;

void solve() {
    int n;
    long long k;
    cin >> n >> k;
    
    vector<long long> a(n);
    map<long long, long long> rem_counts;
    
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        long long rem = (k - (a[i] % k)) % k;
        if (rem != 0) {
            rem_counts[rem]++;
        }
    }
    
    // If no elements need modification, 0 moves are required
    if (rem_counts.empty()) {
        cout << 0 << "\n";
        return;
    }
    
    long long max_moves = 0;
    
    // Iterate through the map to find the remainder that requires the most steps
    for (auto const& [rem, count] : rem_counts) {
        long long current_needed_x = rem + (count - 1) * k;
        max_moves = max(max_moves, current_needed_x);
    }
    
    // Since x starts at 0, the number of moves is max_x + 1
    cout << max_moves + 1 << "\n";
}

int main() {
    // Fast I/O
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    
    int t;
    cin >> t;
    while (t--) {
        solve();
    }
    
    return 0;
}
